foreach(stage IN LISTS SPSC_STAGES)
    get_target_property(lto ${stage} INTERPROCEDURAL_OPTIMIZATION)
    foreach(tool IN ITEMS stress_test)
        add_executable(${tool}_${stage} ${tool}.cpp)
        target_link_libraries(${tool}_${stage} PRIVATE ${stage} Threads::Threads)
        set_target_properties(${tool}_${stage} PROPERTIES INTERPROCEDURAL_OPTIMIZATION ${lto})
    endforeach()
    add_test(NAME stress_${stage} COMMAND stress_test_${stage})
endforeach()
